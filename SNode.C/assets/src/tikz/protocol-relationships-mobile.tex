\begin{tikzpicture}[mqtt mobile]

% Keep the HTTP-specific relationships compact while avoiding any semantic
% container that could imply persistent HTTP containment after upgrade.
\node[mqtt process] (sse)
  {SSE\\EventSource\\within HTTP};
\node[mqtt process,right=\MQHorizontalGapTight of sse] (ws)
  {WebSocket\\via HTTP\\Upgrade};
\path (sse.south) -- (ws.south) coordinate[midway] (branchbottom);
\path (sse.north) -- (ws.north) coordinate[midway] (branchtop);

\node[mqtt state,below=\MQConnectorSpan of branchbottom] (http) {HTTP context};
\node[mqtt process,above=\MQConnectorSpan of ws,mqtt node compact] (mqttws)
  {MQTT over\\WebSocket\\subprotocol};
\path (branchtop |- mqttws.north) coordinate (expressbase);
\node[mqtt application,above=\MQConnectorSpan of expressbase,mqtt node compact] (express)
  {Express-style routing\\and middleware};

\MQDistributedPortNorth{http}{http-n1}{1}{3}
\MQDistributedPortNorth{http}{http-n2}{2}{3}
\MQDistributedPortNorth{http}{http-n3}{3}{3}
\draw[mqtt association] (express.south) -- (http-n2);
\path (http-n1) -- (sse.south) coordinate[midway] (ssemid);
\draw[mqtt association] (http-n1) -- (http-n1 |- ssemid) -- (sse.south |- ssemid) -- (sse.south);
\path (http-n3) -- (ws.south) coordinate[midway] (wsmid);
\draw[mqtt handoff] (http-n3) -- (http-n3 |- wsmid) -- (ws.south |- wsmid) -- (ws.south);
\draw[mqtt association] (ws.north) -- (mqttws.south);

% The two other direct stream roots form a balanced pair below the expanded HTTP
% relationship hub. The HTTP-to-stream association passes between them.
\path (http.south) ++(0,-\MQSectionSeparation) coordinate (directmid);
\node[mqtt process,left=\MQHorizontalGapTight of directmid,mqtt node compact] (custom)
  {custom byte protocol\\\texttt{SocketContext}};
\node[mqtt process,right=\MQHorizontalGapTight of directmid,mqtt node compact] (mqttdirect)
  {direct\\MQTT 3.1.1};
\path (custom.south) -- (mqttdirect.south) coordinate[midway] (directbottom);

\node[mqtt state,below=\MQSectionToControlGap of directbottom,mqtt node extra wide] (stream)
  {established stream connection};
\draw[mqtt association] (custom.south) -- (stream.north -| custom.south);
\draw[mqtt association] (http.south) -- (stream.north);
\draw[mqtt association] (mqttdirect.south) -- (stream.north -| mqttdirect.south);

\node[mqtt layout fit,fit=(express)(mqttws)] (topfit) {};
\node[mqtt section title,above=\MQTitleGap of topfit] (title)
  {Protocol / application\\relationships};

\begin{scope}[on background layer]
  \node[mqtt figure frame,fit=(title)(express)(mqttws)(sse)(ws)(http)(custom)(mqttdirect)(stream)] {};
\end{scope}

\end{tikzpicture}
