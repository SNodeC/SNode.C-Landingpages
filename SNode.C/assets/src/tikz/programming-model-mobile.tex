\node[mqtt layout fit,fit=(server)(serverconn)(serverfactory)(serverctx)] (servercontent) {};

\draw[mqtt flow] (serverfactory.west) -- node[mqtt edge label below] {returns new} (serverctx.east);

\node[mqtt process,below=\MQGapXL of loop,mqtt node compact] (client)

\draw[mqtt flow] (clientfactory.west) -- node[mqtt edge label below] {returns new} (clientctx.east);

\node[mqtt group,fit=(servertitle)(server)(serverconn)(serverfactory)(serverctx)] (servergroup) {};
\node[mqtt group,fit=(clienttitle)(client)(clientconn)(clientfactory)(clientctx)] (clientgroup) {};
