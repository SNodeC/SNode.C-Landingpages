\documentclass[tikz,border=8pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\input{landingpages-figure-system.tex}
\begin{document}
\begin{tikzpicture}[mqtt mobile]

\node[mqtt section title] (retrytitle) {Retry};
\node[mqtt process,below=\MQGapS of retrytitle,mqtt node wide] (attempt)
  {listen / connect\\attempt};
\node[mqtt error,below=\MQConnectorSpan of attempt,mqtt node wide] (failed)
  {establishment fails};
\node[mqtt decision,below=\MQConnectorSpan of failed] (retrypolicy) {retry allowed?};
\node[mqtt process,below=\MQConnectorSpan of retrypolicy,mqtt node wide] (backoff)
  {wait / backoff};
\node[mqtt process,below=\MQConnectorSpan of backoff,mqtt node wide] (newattempt)
  {new listen / connect\\attempt};

\path (retrypolicy.west |- newattempt.south) coordinate (retryendanchor);
\node[mqtt process,below=\MQConnectorSpan of retryendanchor,anchor=north west] (retryend)
  {flow\\ends};
\path (backoff.west) ++(-\MQHorizontalGapTight,0) coordinate (retrynorailx);
\path (retrynorailx |- retrypolicy.west) coordinate (retrynorailtop);
\path (retrynorailtop |- retryend.west) coordinate (retrynorailbottom);
\path (attempt.east) ++(\MQHorizontalGap,0) coordinate (repeatrailtop);
\path (repeatrailtop |- newattempt.east) coordinate (repeatrailbottom);

\draw[mqtt control flow] (attempt.south) -- (failed.north);
\draw[mqtt control flow] (failed.south) -- (retrypolicy.north);
\draw[mqtt control flow] (retrypolicy.south) -- node[mqtt branch label right] {yes} (backoff.north);
\draw[mqtt control flow] (retrypolicy.west)
  -- node[mqtt branch label above] {no} (retrynorailtop)
  -- (retrynorailbottom)
  -- (retryend.west);
\draw[mqtt control flow] (backoff.south) -- (newattempt.north);
\draw[mqtt control flow] (newattempt.east)
  -- (repeatrailbottom)
  -- (repeatrailtop)
  -- node[mqtt edge label above] {repeat} (attempt.east);
\node[mqtt layout fit,fit=(retrytitle)(attempt)(failed)(retrypolicy)(backoff)(newattempt)(retryend)(retrynorailtop)(retrynorailbottom)(repeatrailtop)(repeatrailbottom)] (retryfit) {};

\node[mqtt section title,below=\MQSectionSeparation of retryfit] (reconnecttitle) {Reconnect};
\node[mqtt success,below=\MQGapS of reconnecttitle,mqtt node wide] (connected)
  {connection established};
\node[mqtt error,below=\MQConnectorSpan of connected,mqtt node wide] (interrupted)
  {connection interrupted};
\node[mqtt decision,below=\MQConnectorSpan of interrupted] (reconnectpolicy)
  {reconnect enabled?};
\node[mqtt process,below=\MQConnectorSpan of reconnectpolicy,mqtt node wide] (delay)
  {reconnect delay};
\node[mqtt process,below=\MQConnectorSpan of delay,mqtt node wide] (newcycle)
  {new connect cycle};

\path (reconnectpolicy.west |- newcycle.south) coordinate (reconnectendanchor);
\node[mqtt process,below=\MQConnectorSpan of reconnectendanchor,anchor=north west] (reconnectend)
  {flow\\ends};
\path (delay.west) ++(-\MQHorizontalGapTight,0) coordinate (reconnectnorailx);
\path (reconnectnorailx |- reconnectpolicy.west) coordinate (reconnectnorailtop);
\path (reconnectnorailtop |- reconnectend.west) coordinate (reconnectnorailbottom);

\draw[mqtt control flow] (connected.south) -- (interrupted.north);
\draw[mqtt control flow] (interrupted.south) -- (reconnectpolicy.north);
\draw[mqtt control flow] (reconnectpolicy.south) -- node[mqtt branch label right] {yes} (delay.north);
\draw[mqtt control flow] (reconnectpolicy.west)
  -- node[mqtt branch label above] {no} (reconnectnorailtop)
  -- (reconnectnorailbottom)
  -- (reconnectend.west);
\draw[mqtt control flow] (delay.south) -- (newcycle.north);
\node[mqtt layout fit,fit=(reconnecttitle)(connected)(interrupted)(reconnectpolicy)(delay)(newcycle)(reconnectend)(reconnectnorailtop)(reconnectnorailbottom)] (reconnectfit) {};

\node[mqtt note,below=\MQSectionToControlGap of reconnectfit,mqtt node extra wide] (relationship)
  {Reconnect starts a new client connect cycle.\\If its establishment fails,\\retry policy can apply there.};

\begin{scope}[on background layer]
  \node[mqtt figure frame,fit=(retryfit)(reconnectfit)(relationship)] {};
  \node[mqtt group,fit=(retrytitle)(attempt)(failed)(retrypolicy)(backoff)(newattempt)(retryend)(retrynorailtop)(retrynorailbottom)(repeatrailtop)(repeatrailbottom)] (retrygroup) {};
  \node[mqtt group,fit=(reconnecttitle)(connected)(interrupted)(reconnectpolicy)(delay)(newcycle)(reconnectend)(reconnectnorailtop)(reconnectnorailbottom)] (reconnectgroup) {};
\end{scope}

\end{tikzpicture}
\end{document}
