\begin{tikzpicture}[mqtt desktop]

\node[mqtt config,mqtt node compact] (api)
  {1 --- API and defaults\\lowest precedence};
\node[mqtt config,right=\MQHorizontalGap of api,mqtt node compact] (file)
  {2 --- configuration file\\middle precedence};
\node[mqtt config,right=\MQHorizontalGap of file,mqtt node compact] (cli)
  {3 --- command line\\highest precedence};
\node[mqtt control,below=\MQConnectorSpan of file,mqtt node wide] (resolve) {value resolution};
\node[mqtt state,below=\MQConnectorSpan of resolve,mqtt node wide] (effective) {effective configuration};

\MQDistributedPortNorth{resolve}{resolve-n1}{1}{3}
\MQDistributedPortNorth{resolve}{resolve-n2}{2}{3}
\MQDistributedPortNorth{resolve}{resolve-n3}{3}{3}
\path (api.south) -- (resolve-n1) coordinate[midway] (apimid);
\draw[mqtt flow] (api.south) -- (api.south |- apimid) -- (resolve-n1 |- apimid) -- (resolve-n1);
\draw[mqtt flow] (file.south) -- (resolve-n2);
\path (cli.south) -- (resolve-n3) coordinate[midway] (climid);
\draw[mqtt flow] (cli.south) -- (cli.south |- climid) -- (resolve-n3 |- climid) -- (resolve-n3);
\draw[mqtt control flow] (resolve.south) -- (effective.north);

\node[mqtt section title,below=\MQSectionSeparation of effective] (outputstitle)
  {Effective-value inspection / persistence};
\node[mqtt process,below=\MQGapS of outputstitle,mqtt node compact] (cmd)
  {\texttt{--command-line=...}};
\node[mqtt process,left=\MQHorizontalGapTight of cmd,mqtt node compact] (show)
  {\texttt{--show-config}};
\node[mqtt process,right=\MQHorizontalGapTight of cmd,mqtt node compact] (write)
  {\texttt{--write-config}\\persistent values};
\node[mqtt layout fit,fit=(outputstitle)(show)(cmd)(write)] (outputsfit) {};
\node[mqtt note,below=\MQGapXL of outputsfit,mqtt node extra wide] (help)
  {\texttt{--help=expanded} inspects the hierarchy};

\begin{scope}[on background layer]
  \node[mqtt figure frame,fit=(api)(file)(cli)(resolve)(effective)(outputsfit)(help)] {};
  \node[mqtt group,fit=(outputstitle)(show)(cmd)(write)] (outputs) {};
\end{scope}

\draw[mqtt observation] (effective.south) -- node[mqtt edge label right] {inspect / persist} (outputs.north);

\end{tikzpicture}
