\documentclass[tikz,border=8pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\input{landingpages-figure-system.tex}
\begin{document}
\begin{tikzpicture}[mqtt desktop]

\node[mqtt process,mqtt node compact] (attempt) {listen / connect attempt};
\node[mqtt section title,above=\MQGapS of attempt] (retrytitle) {Retry};
\node[mqtt error,below=\MQConnectorSpan of attempt,mqtt node compact] (failed)
  {establishment fails};
\node[mqtt decision,below=\MQConnectorSpan of failed] (retrypolicy) {retry allowed?};
\node[mqtt process,below=\MQConnectorSpan of retrypolicy,mqtt node compact] (backoff)
  {wait / backoff};
\node[mqtt process,below=\MQConnectorSpan of backoff,mqtt node compact] (newattempt)
  {new listen / connect attempt};

\node[mqtt process,below=\MQConnectorSpan of newattempt] (retryend) {flow ends};
\path (attempt.west) ++(-\MQHorizontalGapTight,0) coordinate (retrynorailx);
\path (retrynorailx |- retrypolicy.west) coordinate (retrynorailtop);
\path (retrynorailtop |- retryend.west) coordinate (retrynorailbottom);
\path (attempt.east) ++(\MQHorizontalGapTight,0) coordinate (repeatrailx);
\path (repeatrailx |- attempt.east) coordinate (repeatrailtop);
\path (repeatrailtop |- newattempt.east) coordinate (repeatrailbottom);

\draw[mqtt control flow] (attempt.south) -- (failed.north);
\draw[mqtt control flow] (failed.south) -- (retrypolicy.north);
\draw[mqtt control flow] (retrypolicy.south) -- node[mqtt branch label right] {yes} (backoff.north);
\draw[mqtt control flow] (retrypolicy.west)
  -- node[mqtt branch label above] {no} (retrynorailtop)
  -- (retrynorailbottom)
  -- (retryend.west);
\draw[mqtt control flow] (backoff.south) -- (newattempt.north);
\draw[mqtt control flow] (newattempt.east)
  -- (repeatrailbottom)
  -- (repeatrailtop)
  -- (attempt.east);

\node[mqtt success,right=\MQSectionSeparation of attempt,mqtt node compact] (connected)
  {connection established};
\node[mqtt section title,above=\MQGapS of connected] (reconnecttitle) {Reconnect};
\node[mqtt error,below=\MQConnectorSpan of connected,mqtt node compact] (interrupted)
  {connection interrupted};
\node[mqtt decision,below=\MQConnectorSpan of interrupted] (reconnectpolicy)
  {reconnect enabled?};
\node[mqtt process,below=\MQConnectorSpan of reconnectpolicy,mqtt node compact] (delay)
  {reconnect delay};
\node[mqtt process,below=\MQConnectorSpan of delay,mqtt node compact] (newcycle)
  {new connect cycle};

\node[mqtt process,below=\MQConnectorSpan of newcycle] (reconnectend) {flow ends};
\path (interrupted.east) ++(\MQHorizontalGapTight,0) coordinate (reconnectnorailx);
\path (reconnectnorailx |- reconnectpolicy.east) coordinate (reconnectnorailtop);
\path (reconnectnorailtop |- reconnectend.east) coordinate (reconnectnorailbottom);
\path (interrupted.west) ++(-\MQHorizontalGapTight,0) coordinate (reconnectbalanceleft);

\draw[mqtt control flow] (connected.south) -- (interrupted.north);
\draw[mqtt control flow] (interrupted.south) -- (reconnectpolicy.north);
\draw[mqtt control flow] (reconnectpolicy.south) -- node[mqtt branch label right] {yes} (delay.north);
\draw[mqtt control flow] (reconnectpolicy.east)
  -- node[mqtt branch label above] {no} (reconnectnorailtop)
  -- (reconnectnorailbottom)
  -- (reconnectend.east);
\draw[mqtt control flow] (delay.south) -- (newcycle.north);

\node[mqtt group,fit=(retrytitle)(attempt)(failed)(retrypolicy)(backoff)(newattempt)(retryend)(retrynorailtop)(retrynorailbottom)(repeatrailtop)(repeatrailbottom)] (retrygroup) {};
\node[mqtt group,fit=(reconnecttitle)(connected)(interrupted)(reconnectpolicy)(delay)(newcycle)(reconnectend)(reconnectnorailtop)(reconnectnorailbottom)(reconnectbalanceleft)] (reconnectgroup) {};
\node[mqtt layout fit,fit=(retrygroup)(reconnectgroup)] (lanesfit) {};
\node[mqtt note,mqtt text wide,below=\MQSectionToControlGap of lanesfit] (relationship)
  {Retry applies while establishing listen/connect. Reconnect is client policy after an established connection is lost. Reconnect starts a new connect cycle; establishment failures there can then use retry policy.};

\begin{scope}[on background layer]
  \node[mqtt figure frame,fit=(retrygroup)(reconnectgroup)(relationship)] {};
\end{scope}

\end{tikzpicture}
\end{document}
