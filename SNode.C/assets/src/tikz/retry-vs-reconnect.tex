\documentclass[tikz,border=8pt]{standalone}
\usepackage[T1]{fontenc}

\begin{document}
\begin{tikzpicture}[mqtt desktop]

\node[mqtt process,minimum width=38mm] (attempt) {listen / connect attempt};
\node[mqtt section title,above=\MQGapS of attempt] (retrytitle) {Retry --- establishment failure};
\node[mqtt error,below=\MQGapM of attempt,minimum width=38mm] (failed) {establishment fails};
\node[mqtt decision,below=\MQGapM of failed,minimum width=34mm,text width=28mm] (retrypolicy) {retry allowed?};
\node[mqtt process,below=\MQGapM of retrypolicy,minimum width=38mm] (backoff) {wait / backoff};
\node[mqtt process,below=\MQGapM of backoff,minimum width=38mm] (newattempt) {new listen / connect attempt};
\node[mqtt note,left=\MQGapS of retrypolicy,minimum width=17mm,text width=11mm] (retryend) {flow ends};
\path (newattempt.west) ++(-\MQGapM,0) coordinate (retryrail);

\draw[mqtt control flow] (attempt.south) -- (failed.north);
\draw[mqtt control flow] (failed.south) -- (retrypolicy.north);
\draw[mqtt control flow] (retrypolicy.south) -- node[mqtt branch label,right] {yes} (backoff.north);
\draw[mqtt control flow] (retrypolicy.west) -- node[mqtt branch label,above] {no} (retryend.east);
\draw[mqtt control flow] (backoff.south) -- (newattempt.north);
\draw[mqtt control flow] (newattempt.west) -- (retryrail) |- node[mqtt edge label,near end,left] {repeat} (attempt.west);

\node[mqtt success,right={\MQGapS+2*\MQGroupPad} of attempt,minimum width=38mm] (connected) {connection established};
\node[mqtt section title,above=\MQGapS of connected] (reconnecttitle) {Reconnect --- later interruption};
\node[mqtt error,below=\MQGapM of connected,minimum width=38mm] (interrupted) {connection interrupted};
\node[mqtt decision,below=\MQGapM of interrupted,minimum width=34mm,text width=28mm] (reconnectpolicy) {reconnect enabled?};
\node[mqtt process,below=\MQGapM of reconnectpolicy,minimum width=38mm] (delay) {reconnect delay};
\node[mqtt process,below=\MQGapM of delay,minimum width=38mm] (newcycle) {new connect cycle};
\node[mqtt note,right=\MQGapS of reconnectpolicy,minimum width=17mm,text width=11mm] (reconnectend) {flow ends};

\draw[mqtt control flow] (connected.south) -- (interrupted.north);
\draw[mqtt control flow] (interrupted.south) -- (reconnectpolicy.north);
\draw[mqtt control flow] (reconnectpolicy.south) -- node[mqtt branch label,right] {yes} (delay.north);
\draw[mqtt control flow] (reconnectpolicy.east) -- node[mqtt branch label,above] {no} (reconnectend.west);
\draw[mqtt control flow] (delay.south) -- (newcycle.north);

\path (newattempt.south) -- (newcycle.south) coordinate[midway] (lanebottom);
\node[mqtt note,below={\MQGapL+\MQGroupPad} of lanebottom,minimum width=108mm,text width=98mm] (relationship)
  {Retry applies while establishing listen/connect. Reconnect is client policy after an established connection is lost. Reconnect starts a new connect cycle; establishment failures there can then use retry policy.};
\path (retrytitle.north) ++(0,\MQGroupPad) coordinate (outertop);
\path (retryend.west) ++(-\MQGroupPad,0) coordinate (outerleft);
\path (reconnectend.east) ++(\MQGroupPad,0) coordinate (outerright);

\begin{scope}[on background layer]
  \node[mqtt figure frame,fit=(outertop)(outerleft)(outerright)(retrytitle)(attempt)(failed)(retrypolicy)(backoff)(newattempt)(retryend)(retryrail)(reconnecttitle)(connected)(interrupted)(reconnectpolicy)(delay)(newcycle)(reconnectend)(relationship)] {};
  \node[mqtt group,fit=(retrytitle)(attempt)(failed)(retrypolicy)(backoff)(newattempt)(retryend)(retryrail)] (retrygroup) {};
  \node[mqtt group,fit=(reconnecttitle)(connected)(interrupted)(reconnectpolicy)(delay)(newcycle)(reconnectend)] (reconnectgroup) {};
\end{scope}

\end{tikzpicture}
\end{document}
