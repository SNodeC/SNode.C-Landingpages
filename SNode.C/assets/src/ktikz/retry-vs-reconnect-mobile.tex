\begin{tikzpicture}[mqtt mobile]

\node[mqtt section title] (retrytitle) {Retry};
\node[mqtt error,below=\MQGapS of retrytitle,mqtt node wide] (failed)
  {listen / connect\\establishment fails};
\node[mqtt decision,mqtt node compact,below=\MQConnectorSpan of failed] (retrypolicy) {retry?};
\node[mqtt process,below=\MQConnectorSpan of retrypolicy,mqtt node wide] (retryaction)
  {wait / backoff, then\\new listen / connect attempt};
\node[mqtt process,below=\MQConnectorSpan of retryaction,mqtt node compact] (retryend) {flow ends};

\path (retryaction.west) ++(-\MQHorizontalGapTight,0) coordinate (retrynorailx);
\path (retrynorailx |- retrypolicy.west) coordinate (retrynorailtop);
\path (retrynorailtop |- retryend.west) coordinate (retrynorailbottom);
\path (retryaction.east) ++(\MQHorizontalGapTight,0) coordinate (repeatrailx);
\path (repeatrailx |- failed.east) coordinate (repeatrailtop);
\path (repeatrailtop |- retryaction.east) coordinate (repeatrailbottom);

\draw[mqtt control flow] (failed.south) -- (retrypolicy.north);
\draw[mqtt control flow] (retrypolicy.south) -- node[mqtt branch label right] {yes} (retryaction.north);
\draw[mqtt control flow] (retrypolicy.west)
  -- node[mqtt branch label above] {no} (retrynorailtop)
  -- (retrynorailbottom)
  -- (retryend.west);
\draw[mqtt control flow] (retryaction.east)
  -- (repeatrailbottom)
  -- (repeatrailtop)
  -- (failed.east);
\node[mqtt layout fit,fit=(retrytitle)(failed)(retrypolicy)(retryaction)(retryend)(retrynorailtop)(retrynorailbottom)(repeatrailtop)(repeatrailbottom)] (retryfit) {};

\node[mqtt section title,below=\MQMobileLaneGap of retryfit] (reconnecttitle) {Reconnect};
\node[mqtt error,below=\MQGapS of reconnecttitle,mqtt node wide] (interrupted)
  {established connection\\interrupted};
\node[mqtt decision,mqtt node compact,below=\MQConnectorSpan of interrupted] (reconnectpolicy)
  {reconnect?};
\node[mqtt process,below=\MQConnectorSpan of reconnectpolicy,mqtt node wide] (reconnectaction)
  {reconnect delay, then\\new client connect cycle};
\node[mqtt process,below=\MQConnectorSpan of reconnectaction,mqtt node compact] (reconnectend) {flow ends};

\path (reconnectaction.west) ++(-\MQHorizontalGapTight,0) coordinate (reconnectnorailx);
\path (reconnectnorailx |- reconnectpolicy.west) coordinate (reconnectnorailtop);
\path (reconnectnorailtop |- reconnectend.west) coordinate (reconnectnorailbottom);
\path (reconnectaction.east) ++(\MQHorizontalGapTight,0) coordinate (reconnectbalanceright);

\draw[mqtt control flow] (interrupted.south) -- (reconnectpolicy.north);
\draw[mqtt control flow] (reconnectpolicy.south) -- node[mqtt branch label right] {yes} (reconnectaction.north);
\draw[mqtt control flow] (reconnectpolicy.west)
  -- node[mqtt branch label above] {no} (reconnectnorailtop)
  -- (reconnectnorailbottom)
  -- (reconnectend.west);
\node[mqtt layout fit,fit=(reconnecttitle)(interrupted)(reconnectpolicy)(reconnectaction)(reconnectend)(reconnectnorailtop)(reconnectnorailbottom)(reconnectbalanceright)] (reconnectfit) {};

\node[mqtt group,fit=(retrytitle)(failed)(retrypolicy)(retryaction)(retryend)(retrynorailtop)(retrynorailbottom)(repeatrailtop)(repeatrailbottom)] (retrygroup) {};
\node[mqtt group,fit=(reconnecttitle)(interrupted)(reconnectpolicy)(reconnectaction)(reconnectend)(reconnectnorailtop)(reconnectnorailbottom)(reconnectbalanceright)] (reconnectgroup) {};
\node[mqtt note,below=\MQSectionToControlGap of reconnectgroup,mqtt node extra wide] (relationship)
  {If that new connect cycle fails to establish,\\retry policy can apply.};

\begin{scope}[on background layer]
  \node[mqtt figure frame,fit=(retrygroup)(reconnectgroup)(relationship)] {};
\end{scope}

\end{tikzpicture}
