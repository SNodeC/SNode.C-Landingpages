cmake_minimum_required(VERSION 3.18)

project(SNodeCLandingpagesFigures NONE)

# -----------------------------------------------------------------------------
# Reproducible technical-figure generation
# -----------------------------------------------------------------------------
#
# Canonical technical figure sources live below each project:
#
#   <project>/assets/src/tikz/*.tex
#
# Shared canonical visual-system sources live at:
#
#   shared/assets/src/tikz/*-figure-system.tex
#
# Project-local system files remain supported during staged migration and take
# precedence over the shared directory. This keeps the existing MQTTSuite TikZ
# figures unchanged while new SNode.C figures can consume the shared system with
# the same stable \input{...} names.
#
# Figure names are intentionally not enumerated here. New TikZ figures become
# build inputs automatically when their .tex source is added. Shared visual
# system files ending in "-figure-system.tex" are excluded from the figure list
# and are treated as dependencies instead.
#
# Configure/build from the repository root, for example:
#
#   cmake -S . -B cmake-build-figures
#   cmake --build cmake-build-figures --target figures --parallel 8
#
# IMPORTANT: generated SVGs are build outputs, not repository sources. The
# `figures` target depends on one always-run custom target per discovered figure,
# so every invocation recompiles every concrete TikZ source and refreshes its
# local SVG output. CI uploads those generated bytes for review; it never commits
# them back to the repository.

find_program(PDFLATEX_EXECUTABLE NAMES pdflatex REQUIRED)
find_program(PDFTOCAIRO_EXECUTABLE NAMES pdftocairo REQUIRED)
find_program(PDFINFO_EXECUTABLE NAMES pdfinfo REQUIRED)
find_program(SCOUR_EXECUTABLE NAMES scour REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)

# Cairo serializes SVG viewport dimensions as integer PostScript points. Canonical
# technical figures, however, have exact physical publication widths (160 mm desktop,
# 100 mm mobile). First prove that the TeX/PDF source already has the canonical
# physical width; a noncanonical composition must fail instead of being scaled to fit.
# Only then normalize the root SVG physical viewport after scour while preserving
# the generated viewBox coordinate system and therefore all vector geometry.
set(TIKZ_SVG_CANVAS_NORMALIZER "${CMAKE_BINARY_DIR}/normalize-tikz-svg-canvas.py")
file(WRITE "${TIKZ_SVG_CANVAS_NORMALIZER}" [=[
from pathlib import Path
import os
import re
import subprocess
import sys

svg_path = Path(sys.argv[1])
pdf_path = Path(sys.argv[2])
pdfinfo_executable = sys.argv[3]
canvas_width_mm = sys.argv[4]

pdfinfo = subprocess.check_output(
    [pdfinfo_executable, str(pdf_path)],
    text=True,
    env={**os.environ, "LC_ALL": "C"},
)
page_size = re.search(
    r"^Page size:\s+([0-9.]+)\s+x\s+([0-9.]+)\s+pts",
    pdfinfo,
    re.MULTILINE,
)
if page_size is None:
    raise SystemExit(f"Unable to read PDF page size from {pdf_path}")

source_width_pt = float(page_size.group(1))
target_width_pt = float(canvas_width_mm) * 72.0 / 25.4
# pdfinfo reports page dimensions to three decimal places. One millipoint is the
# smallest tolerance that covers that serialization precision without becoming a
# visual/layout allowance.
width_tolerance_pt = 0.001
if abs(source_width_pt - target_width_pt) > width_tolerance_pt:
    raise SystemExit(
        f"Noncanonical technical-figure canvas in {pdf_path}: "
        f"source width {source_width_pt:.3f}pt, expected "
        f"{target_width_pt:.6f}pt ({canvas_width_mm}mm). "
        "Recompose the TikZ source; scaling to the publication width is forbidden."
    )

svg = svg_path.read_text(encoding="utf-8")
root_match = re.search(r'<svg\b[^>]*>', svg)
if root_match is None:
    raise SystemExit(f"Unable to locate SVG root element in {svg_path}")

root = root_match.group(0)
viewbox_match = re.search(
    r'\bviewBox="0 0 ([0-9.]+) ([0-9.]+)"',
    root,
)
if viewbox_match is None:
    raise SystemExit(f"Unable to locate canonical SVG viewBox in {svg_path}")

viewbox_width = float(viewbox_match.group(1))
viewbox_height = float(viewbox_match.group(2))
height_mm = float(canvas_width_mm) * viewbox_height / viewbox_width

# scour versions differ in whether --enable-viewboxing retains width/height and
# in root-attribute ordering. Remove any existing physical viewport and append the
# canonical one without depending on either behavior.
root = re.sub(r'\s+width="[^"]*"', '', root)
root = re.sub(r'\s+height="[^"]*"', '', root)
root = root[:-1] + f' width="{canvas_width_mm}mm" height="{height_mm:.6f}mm">'
svg = svg[:root_match.start()] + root + svg[root_match.end():]
svg_path.write_text(svg, encoding="utf-8")
]=])

set(TIKZ_SHARED_SYSTEM_DIR "${CMAKE_SOURCE_DIR}/shared/assets/src/tikz")
file(
    GLOB TIKZ_SHARED_SYSTEM_DEPENDENCIES
    CONFIGURE_DEPENDS
    "${TIKZ_SHARED_SYSTEM_DIR}/*-figure-system.tex"
)

file(
    GLOB_RECURSE TIKZ_TEX_SOURCES
    CONFIGURE_DEPENDS
    RELATIVE "${CMAKE_SOURCE_DIR}"
    "*/assets/src/tikz/*.tex"
)

set(TIKZ_FIGURE_OUTPUTS "")
set(TIKZ_FIGURE_TARGETS "")
set(TIKZ_FIGURE_NAMES "")

foreach(TIKZ_SOURCE_REL IN LISTS TIKZ_TEX_SOURCES)
    get_filename_component(TIKZ_FILENAME "${TIKZ_SOURCE_REL}" NAME)

    # Shared style/system sources are included by figures but are not figures.
    if(TIKZ_FILENAME MATCHES "-figure-system\\.tex$")
        continue()
    endif()

    # Extract the owning project from
    # <project>/assets/src/tikz/<possibly/nested/path>/<name>.tex.
    if(NOT TIKZ_SOURCE_REL MATCHES "^([^/]+)/assets/src/tikz/(.+)\\.tex$")
        message(FATAL_ERROR
            "Unexpected TikZ figure path: ${TIKZ_SOURCE_REL}. "
            "Expected <project>/assets/src/tikz/<name>.tex"
        )
    endif()

    set(TIKZ_PROJECT "${CMAKE_MATCH_1}")
    set(TIKZ_FIGURE_SUBPATH "${CMAKE_MATCH_2}")

    get_filename_component(TIKZ_SOURCE_ABS
        "${CMAKE_SOURCE_DIR}/${TIKZ_SOURCE_REL}" ABSOLUTE)
    get_filename_component(TIKZ_SOURCE_DIR
        "${TIKZ_SOURCE_ABS}" DIRECTORY)
    get_filename_component(TIKZ_STEM
        "${TIKZ_SOURCE_ABS}" NAME_WE)

    # Generated assets are intentionally flat within each project's assets/
    # directory. Reject duplicate output names early instead of silently
    # overwriting one figure with another.
    set(TIKZ_OUTPUT
        "${CMAKE_SOURCE_DIR}/${TIKZ_PROJECT}/assets/${TIKZ_STEM}.svg")
    list(FIND TIKZ_FIGURE_OUTPUTS "${TIKZ_OUTPUT}" TIKZ_DUPLICATE_INDEX)
    if(NOT TIKZ_DUPLICATE_INDEX EQUAL -1)
        message(FATAL_ERROR
            "Duplicate TikZ output '${TIKZ_OUTPUT}'. "
            "Figure basenames must be unique within one project."
        )
    endif()

    # Give each source its own temporary directory so TeX auxiliary files never
    # collide when all figures build in parallel.
    string(MAKE_C_IDENTIFIER
        "${TIKZ_PROJECT}_${TIKZ_FIGURE_SUBPATH}" TIKZ_WORK_ID)
    set(TIKZ_WORK_DIR
        "${CMAKE_BINARY_DIR}/tikz/${TIKZ_WORK_ID}")
    set(TIKZ_PDF
        "${TIKZ_WORK_DIR}/${TIKZ_STEM}.pdf")
    set(TIKZ_RAW_SVG
        "${TIKZ_WORK_DIR}/${TIKZ_STEM}-raw.svg")
    set(TIKZ_TARGET "tikz_${TIKZ_WORK_ID}")

    if(TIKZ_STEM MATCHES "-mobile$")
        set(TIKZ_PUBLICATION_WIDTH_MM "100")
    else()
        set(TIKZ_PUBLICATION_WIDTH_MM "160")
    endif()

    file(
        GLOB TIKZ_LOCAL_SYSTEM_DEPENDENCIES
        CONFIGURE_DEPENDS
        "${TIKZ_SOURCE_DIR}/*-figure-system.tex"
    )

    # A custom target is intentionally always out of date. Therefore every
    # build of the aggregate `figures` target executes this full TikZ -> PDF ->
    # SVG pipeline and refreshes the untracked generated output.
    add_custom_target(
        "${TIKZ_TARGET}"
        COMMAND "${CMAKE_COMMAND}" -E make_directory "${TIKZ_WORK_DIR}"
        COMMAND
            "${CMAKE_COMMAND}" -E env
            "TEXINPUTS=${TIKZ_SOURCE_DIR}:${TIKZ_SHARED_SYSTEM_DIR}:"
            "${PDFLATEX_EXECUTABLE}"
            -interaction=nonstopmode
            -halt-on-error
            -file-line-error
            -output-directory=${TIKZ_WORK_DIR}
            "${TIKZ_SOURCE_ABS}"
        # Poppler's Cairo SVG backend renders the PDF glyphs directly to vector
        # paths. This keeps the exact TeX appearance without relying on fonts in
        # the reader's browser and avoids Inkscape PDF-import CLI/version and
        # ligature-loss problems observed with Ubuntu's Inkscape 1.2.2.
        COMMAND
            "${PDFTOCAIRO_EXECUTABLE}"
            -svg
            "${TIKZ_PDF}"
            "${TIKZ_RAW_SVG}"
        COMMAND
            "${SCOUR_EXECUTABLE}"
            -i "${TIKZ_RAW_SVG}"
            -o "${TIKZ_OUTPUT}"
            --enable-viewboxing
            --enable-id-stripping
            --enable-comment-stripping
            --shorten-ids
            --strip-xml-prolog
            --indent=none
            --set-precision=3
            --set-c-precision=3
        COMMAND
            "${Python3_EXECUTABLE}"
            "${TIKZ_SVG_CANVAS_NORMALIZER}"
            "${TIKZ_OUTPUT}"
            "${TIKZ_PDF}"
            "${PDFINFO_EXECUTABLE}"
            "${TIKZ_PUBLICATION_WIDTH_MM}"
        BYPRODUCTS
            "${TIKZ_PDF}"
            "${TIKZ_RAW_SVG}"
            "${TIKZ_OUTPUT}"
        DEPENDS
            "${TIKZ_SOURCE_ABS}"
            ${TIKZ_LOCAL_SYSTEM_DEPENDENCIES}
            ${TIKZ_SHARED_SYSTEM_DEPENDENCIES}
        WORKING_DIRECTORY "${TIKZ_SOURCE_DIR}"
        COMMENT "TikZ -> SVG: ${TIKZ_SOURCE_REL}"
        VERBATIM
    )

    list(APPEND TIKZ_FIGURE_OUTPUTS "${TIKZ_OUTPUT}")
    list(APPEND TIKZ_FIGURE_TARGETS "${TIKZ_TARGET}")
    list(APPEND TIKZ_FIGURE_NAMES "${TIKZ_PROJECT}/${TIKZ_STEM}")
endforeach()

if(TIKZ_FIGURE_TARGETS)
    add_custom_target(figures ALL DEPENDS ${TIKZ_FIGURE_TARGETS})

    list(LENGTH TIKZ_FIGURE_TARGETS TIKZ_FIGURE_COUNT)
    message(STATUS "Discovered ${TIKZ_FIGURE_COUNT} TikZ figure source(s)")
    message(STATUS "The figures target recompiles every discovered source on every build")
    foreach(TIKZ_FIGURE_NAME IN LISTS TIKZ_FIGURE_NAMES)
        message(STATUS "  figure: ${TIKZ_FIGURE_NAME}")
    endforeach()
else()
    # Keep the target available before the first concrete figure source lands.
    add_custom_target(figures)
    message(STATUS "No TikZ figure sources found yet")
endif()
