\documentclass[tikz,border=0pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\input{mqttsystem-figure-system.tex}
\input{snodec-canonical-figure-system.tex}
\begin{document}
\begin{tikzpicture}[mqtt desktop]
\tikzset{
  input/.style={mqtt external,minimum width=58mm,text width=52mm},
  dispatcher/.style={mqtt control,minimum width=76mm,text width=70mm},
  mode/.style={mqtt control,minimum width=42mm,text width=36mm,minimum height=23mm},
  collector/.style={mqtt process,minimum width=76mm,text width=70mm},
  result/.style={mqtt process,minimum width=76mm,text width=70mm},
  foot/.style={mqtt note,minimum width=128mm,text width=120mm}
}
\def\MQArrowSpan{8.3mm}
\def\MQHalfArrowSpan{4.15mm}

\node[input] (input)
  {matched MQTT PUBLISH\\topic + payload + metadata};
\node[dispatcher,below=\MQArrowSpan of input] (dispatch)
  {process every configured mapping key present\\one subscription may contain \texttt{static}, \texttt{value}, and \texttt{json} together};
\draw[mqtt control flow] (input.south) -- (dispatch.north);

\node[mode,below=\MQArrowSpan of dispatch] (value)
  {\textbf{value}\\raw payload becomes template \texttt{message}\\object or array → 0..n outputs};
\node[mode,left=\MQArrowSpan of value] (static)
  {\textbf{static}\\exact payload-string lookup\\object or array → 0..n outputs};
\node[mode,right=\MQArrowSpan of value] (json)
  {\textbf{json}\\parse JSON; parsed value becomes \texttt{message}\\object or array → 0..n outputs};

\coordinate (dispatchStaticStart) at ($(dispatch.south west)!.25!(dispatch.south east)$);
\coordinate (dispatchStaticMid) at ($(dispatchStaticStart)+(0,-\MQHalfArrowSpan)$);
\draw[mqtt control flow]
  (dispatchStaticStart) -- (dispatchStaticMid)
  -- (static.north |- dispatchStaticMid) -- (static.north);
\draw[mqtt control flow] (dispatch.south) -- (value.north);
\coordinate (dispatchJsonStart) at ($(dispatch.south west)!.75!(dispatch.south east)$);
\coordinate (dispatchJsonMid) at ($(dispatchJsonStart)+(0,-\MQHalfArrowSpan)$);
\draw[mqtt control flow]
  (dispatchJsonStart) -- (dispatchJsonMid)
  -- (json.north |- dispatchJsonMid) -- (json.north);

\node[collector,below=\MQArrowSpan of value] (collector)
  {append every produced mapping to the common result\\mapped topic + payload + QoS + retain + delay};

\coordinate (staticCollectTarget) at ($(collector.north west)!.25!(collector.north east)$);
\coordinate (staticCollectMid) at ($(static.south)!0.5!(staticCollectTarget)$);
\draw[mqtt control flow]
  (static.south) -- (static.south |- staticCollectMid)
  -- (staticCollectTarget |- staticCollectMid) -- (staticCollectTarget);
\draw[mqtt control flow] (value.south) -- (collector.north);
\coordinate (jsonCollectTarget) at ($(collector.north west)!.75!(collector.north east)$);
\coordinate (jsonCollectMid) at ($(json.south)!0.5!(jsonCollectTarget)$);
\draw[mqtt control flow]
  (json.south) -- (json.south |- jsonCollectMid)
  -- (jsonCollectTarget |- jsonCollectMid) -- (jsonCollectTarget);

\node[result,below=\MQArrowSpan of collector] (result)
  {combined result\\\textbf{zero / one / many} mapped PUBLISH messages};
\draw[mqtt data flow] (collector.south) -- (result.north);

\node[foot,below=\MQGapM of result] (note)
  {The keys are independent sibling rules, not mutually exclusive stages. Arrays fan out; static no-match, JSON parse/render failure, or suppression can contribute zero outputs.};

\coordinate (titleY) at ($(input.north)+(0,\MQTitleGap)$);
\node[mqtt section title,anchor=south west] (figureTitle)
  at (static.west |- titleY) {Static, scalar/value, JSON, and fan-out mapping};
\begin{pgfonlayer}{background}
  \node[mqtt figure frame,fit=(figureTitle)(input)(dispatch)(static)(value)(json)(collector)(result)(note)] (frame) {};
\end{pgfonlayer}
\end{tikzpicture}
\end{document}
