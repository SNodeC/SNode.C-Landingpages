\node[mqtt state,wide] (active) {active mapping};
\node[mqtt control,wide,below=\MQGapS of active] (edit)
  {PATCH / POST \texttt{/config}};
\node[mqtt config,wide,below=\MQGapS of edit] (draft) {draft mapping};
\node[mqtt note,wide,below=\MQGapS of draft] (validate)
  {optional validation preflight};
\node[mqtt control,wide,below=\MQGapM of validate] (deploy)
  {POST \texttt{/config/deploy}};
\node[mqtt config,wide,below=\MQGapS of deploy] (backup)
  {backup prior active file into local version history};
\node[mqtt process,wide,below=\MQGapS of backup] (apply)
  {\texttt{setMapping()} + attempt active-file persistence};
\node[mqtt control,wide,below=\MQGapM of apply] (reload)
  {connection changed?\\YES → reconnect / DISCONNECT\\NO → subscription delta};

\draw[mqtt control flow] (active.south) -- (edit.north);
\draw[mqtt control flow] (edit.south) -- (draft.north);
\draw[mqtt association] (draft.south) -- (validate.north);
\draw[mqtt control flow] (validate.south) -- (deploy.north);
\draw[mqtt control flow] (deploy.south) -- (backup.north);
\draw[mqtt control flow] (backup.south) -- (apply.north);
\draw[mqtt control flow] (apply.south) -- (reload.north);

\node[mqtt warning,wide,below=\MQGapM of reload] (persist)
  {Persistence is attempted, but its return value is not used as the current deploy/rollback HTTP-success gate.};

\node[mqtt config,wide,below=\MQMobileLaneGap of persist] (history)
  {version history contains prior active files};
\node[mqtt control,wide,below=\MQGapS of history] (rollback)
  {later optional POST \texttt{/config/rollback} restores a selected version};
\node[mqtt note,wide,below=\MQGapS of rollback] (reenter)
  {rollback then uses the same \texttt{setMapping()} / persistence-attempt / reload decision path};
\draw[mqtt control flow] (history.south) -- (rollback.north);
\draw[mqtt dependency] (rollback.south) -- (reenter.north);
\draw[mqtt association] (backup.east) -- ++(8mm,0) |- ($(history.east)+(8mm,0)$) -- (history.east);

  \node[mqtt figure frame,fit=(active)(edit)(draft)(validate)(deploy)(backup)(apply)(reload)(persist)(history)(rollback)(reenter)] (frame) {};

  {history / rollback is optional operator activity, not a mandatory post-deploy stage};
