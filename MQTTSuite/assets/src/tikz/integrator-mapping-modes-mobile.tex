  {\textbf{value}: whole payload becomes template \texttt{message}};
\node[mqtt control,wide,below=\MQGapS of value] (json)
  {\textbf{json}: parse payload as JSON; expose JSON as template \texttt{message}};

\coordinate (inbus-top) at ($(static.west)+(-7mm,0)$);
\coordinate (inbus-mid) at ($(value.west)+(-7mm,0)$);

\draw[mqtt control flow] (inbus-mid) -- (value.west);
\draw[mqtt control flow] (inbus-bottom) -- (json.west);

\node[mqtt process,wide,below=\MQGapM of json] (common)
  {common mapped-output model\\topic + payload + QoS + retain + delay};
\coordinate (outbus-top) at ($(static.east)+(7mm,0)$);
\coordinate (outbus-mid) at ($(value.east)+(7mm,0)$);

\draw[mqtt association] (value.east) -- (outbus-mid);

\draw[mqtt control flow] (outbus-bottom) |- (common.east);

\node[mqtt process,wide,below=\MQGapS of common] (cardinality)
  {\textbf{zero / one / many} mapped PUBLISH messages; arrays provide fan-out};
\draw[mqtt data flow] (common.south) -- (cardinality.north);
\node[mqtt note,wide,below=\MQGapS of cardinality] (note)
  {Modes are not a sequential pipeline. Invalid, suppressed, or unmatched mappings can produce zero outputs.};
