\begin{document}
\begin{tikzpicture}[mqtt desktop]
\tikzset{
  root/.style={minimum width=38mm,text width=32mm,minimum height=13mm},
  inst/.style={minimum width=42mm,text width=36mm,minimum height=13mm},
  peer/.style={mqtt config,minimum width=41mm,text width=35mm,minimum height=19mm},
  side/.style={mqtt note,minimum width=41mm,text width=35mm,minimum height=17mm}
}
\def\OwnGap{8.3mm}

% Ownership: application root -> selected instance; section families inside are peers.
\node[mqtt application,root] (app) at (-25mm,0)
  {\textbf{application root}\\e.g. \texttt{mqttcli}};
\node[mqtt control,inst,below=\OwnGap of app] (instance)
  {\textbf{named connection / listener instance}\\e.g. \texttt{in-mqtt}};
\draw[mqtt contains] (app.south) -- (instance.north);

\node[peer,below=10mm of instance,xshift=-23.5mm] (endpoint)
  {\textbf{endpoint + transport}\\\texttt{remote} / \texttt{local}\\address, socket / TLS options};
\node[peer,right=6mm of endpoint] (web)
  {\textbf{HTTP / WebSocket}\\optional section(s)\\for instance families that provide them};
\node[peer,below=6mm of endpoint] (session)
  {\textbf{MQTT session}\\CONNECT / session settings\\where MQTT is used};
\node[peer,right=6mm of session] (appspecific)
  {\textbf{application-specific}\\e.g. \texttt{sub}, \texttt{pub}\\or \texttt{db} with nested \texttt{storage}};
\node[fit=(instance)(endpoint)(web)(session)(appspecific),inner sep=4.5mm] (instanceBounds) {};

% Configuration surfaces are deliberately separate from the ownership diagram.
\node[side,anchor=north west] (ownership) at ($(instanceBounds.north east)+(5mm,0)$)
  {\textbf{ownership rule}\\Peer section families belong to the selected instance. Exact sets vary by application and instance family. This is configuration ownership, not runtime protocol layering or execution order.};
\node[side,below=6mm of ownership] (values)
  {\textbf{value precedence}\\API / defaults\\$<$ configuration file\\$<$ command line};
\node[side,below=6mm of values] (inspect)
  {\textbf{introspection}\\tree: \texttt{--help=expanded}\\state: \texttt{--show-config}\\\texttt{--command-line=...}};
\node[fit=(values)(inspect),inner sep=4.5mm] (surfaceBounds) {};

\coordinate (leftEdge) at (instanceBounds.west);
\coordinate (titleY) at ($(app.north)+(0,\MQTitleGap)$);
\node[mqtt section title,anchor=south west] (figureTitle)
  at (leftEdge |- titleY) {MQTTSuite configuration ownership hierarchy};

\begin{pgfonlayer}{background}
  \node[mqtt figure frame,fit=(figureTitle)(app)(instanceBounds)(ownership)(surfaceBounds)] {};
  \node[mqtt group,fit=(instance)(endpoint)(web)(session)(appspecific),inner sep=4.5mm] (instanceGroup) {};
  \node[mqtt group,fit=(values)(inspect),inner sep=4.5mm] (surfaceGroup) {};
\end{pgfonlayer}
\node[mqtt edge label,anchor=west] at ($(instanceGroup.north west)+(3mm,0)$)
  {selected instance — peer configuration sections};
\node[mqtt edge label,anchor=west] at ($(surfaceGroup.north west)+(3mm,0)$)
  {configuration surfaces};
\end{tikzpicture}
\end{document}
