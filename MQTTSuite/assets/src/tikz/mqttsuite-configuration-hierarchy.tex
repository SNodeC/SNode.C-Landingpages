\documentclass[tikz,border=0pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\input{mqttsystem-figure-system.tex}
\input{snodec-canonical-figure-system.tex}
\begin{document}
\begin{tikzpicture}[mqtt desktop]
\tikzset{
  root/.style={minimum width=36mm,text width=30mm,minimum height=13mm},
  inst/.style={minimum width=40mm,text width=34mm,minimum height=13mm},
  peer/.style={mqtt config,minimum width=39mm,text width=33mm,minimum height=19mm},
  side/.style={mqtt note,minimum width=39mm,text width=33mm,minimum height=17mm}
}

% Ownership: application root -> selected instance. The section families are peers.
\node[mqtt application,root] (app)
  {\textbf{application root}\\e.g. \texttt{mqttcli}};
\node[mqtt control,inst,below=\MQGapM of app] (instance)
  {\textbf{named connection / listener instance}\\e.g. \texttt{in-mqtt}};
\draw[mqtt contains] (app.south) -- (instance.north);

\node[mqtt annotation,below=\MQLabelGap of instance] (sectionLabel)
  {selected instance — peer configuration sections};
\matrix (peerGrid) [
  matrix of nodes,
  below=\MQGapS of sectionLabel,
  row sep=\MQGapS,
  column sep=\MQHorizontalGapTight,
  nodes={peer,anchor=center}
] {
  {\textbf{endpoint + transport}\\\texttt{remote} / \texttt{local}\\address, socket / TLS options} &
  {\textbf{HTTP / WebSocket}\\optional section(s)\\for instance families that provide them} \\
  {\textbf{MQTT session}\\CONNECT / session settings\\where MQTT is used} &
  {\textbf{application-specific}\\e.g. \texttt{sub}, \texttt{pub}\\or \texttt{db} with nested \texttt{storage}} \\
};
\node[fit=(instance)(sectionLabel)(peerGrid),inner sep=\MQGroupPad] (instanceBounds) {};

% Configuration surfaces stay separate from configuration ownership. Aligning
% this explanatory column with the application root balances the desktop canvas.
\node[side,anchor=north west] (ownership)
  at ($(instanceBounds.east |- app.north)+(\MQHorizontalGapTight,0)$)
  {\textbf{ownership rule}\\Peer section families belong directly to the selected instance. Exact sets vary by application and instance family. Runtime protocol layering and execution order are separate concepts.};

\node[mqtt annotation,below=\MQGapS of ownership] (surfaceLabel)
  {configuration surfaces};
\node[side,below=\MQLabelGap of surfaceLabel] (values)
  {\textbf{value precedence}\\API / defaults\\$<$ configuration file\\$<$ command line};
\node[side,below=\MQGapS of values] (inspect)
  {\textbf{introspection}\\tree: \texttt{--help=expanded}\\state: \texttt{--show-config}\\\texttt{--command-line=...}};
\node[fit=(surfaceLabel)(values)(inspect),inner sep=\MQGroupPad] (surfaceBounds) {};

\coordinate (leftEdge) at (instanceBounds.west);
\coordinate (titleY) at ($(app.north)+(0,\MQTitleGap)$);
\node[mqtt section title,anchor=south west] (figureTitle)
  at (leftEdge |- titleY) {MQTTSuite configuration ownership hierarchy};

\begin{pgfonlayer}{background}
  \node[mqtt figure frame,fit=(figureTitle)(app)(instanceBounds)(ownership)(surfaceBounds)] {};
  \node[mqtt group,fit=(instance)(sectionLabel)(peerGrid),inner sep=\MQGroupPad] {};
  \node[mqtt group,fit=(surfaceLabel)(values)(inspect),inner sep=\MQGroupPad] {};
\end{pgfonlayer}
\end{tikzpicture}
\end{document}
