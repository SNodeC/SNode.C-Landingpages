% Establish the two equal sibling surfaces first; all ingress placement derives
% from their real common top edge, so the branch span is exactly \MQArrowRun.
\node[wssurface] (ws)
  {\textbf{MQTT WebSocket data route}\\\texttt{/ · /ws · /mqtt}\\subprotocol \texttt{mqtt}};
\node[adminsurface,right=\MQHorizontalGap of ws] (admin)
  {\textbf{Operator / browser routes}\\dashboard \texttt{/clients/*}\\JSON \texttt{/api/mqtt/*} · SSE};
\coordinate (surfacesTopMid) at ($(ws.north)!0.5!(admin.north)$);

\node[ingress,above=\MQArrowRun of surfacesTopMid] (router)
  {\textbf{shared Broker HTTP / HTTPS listener + Express router}\\one ingress for browser, administration, SSE, and MQTT WebSocket};
\coordinate (routerWsPort) at ($(router.south west)!.35!(router.south east)$);
\coordinate (routerAdminPort) at ($(router.south west)!.65!(router.south east)$);
\coordinate (wsMid) at ($(routerWsPort)!0.5!(ws.north)$);
\coordinate (adminMid) at ($(routerAdminPort)!0.5!(admin.north)$);
\draw[mqtt data flow]
  (routerWsPort) -- (routerWsPort |- wsMid) -| (ws.north);
\draw[mqtt control flow]
  (routerAdminPort) -- (routerAdminPort |- adminMid) -| (admin.north);

\node[client,above=\MQArrowRun of router] (client)
  {browser / HTTP client or MQTT-WebSocket client};
\draw[mqtt flow] (client.south) -- (router.north);

\begin{pgfonlayer}{background}
  \node[mqtt trust boundary,fit=(router)(ws)(admin)] (trust) {};
\end{pgfonlayer}
\node[mqtt annotation,text width=112mm,align=center,below=\MQLabelGap of trust] (trustLabel)
  {Dashed region: required external / trusted deployment boundary — not enforced by MQTTBroker};

\matrix (securityRow) [matrix of nodes,below=\MQGapM of trustLabel,column sep=\MQHorizontalGap,inner sep=0pt,outer sep=0pt] {
  \node[riskcard] (risk) {\textbf{Current Broker exposure}\\No application authentication on operator routes. Wildcard CORS: \texttt{/api/mqtt/*} and \texttt{/api/mqtt/events}. Client-event JSON/logging can include the supplied MQTT password.}; &
  \node[deploymentcard] (tls) {\textbf{Deployment control}\\HTTPS protects transport, not route authorization. Use loopback/private binding, firewall policy, or authenticated path-aware external access control.}; \\
};

\node[inner sep=0pt,fit=(client)(trust)(trustLabel)(securityRow)] (contentBounds) {};
\coordinate (titleY) at ($(client.north)+(0,\MQTitleGap)$);
\node[mqtt section title,anchor=south west] (figureTitle)
  at (contentBounds.west |- titleY) {MQTTBroker HTTP exposure and required trust boundary};

\begin{pgfonlayer}{framebackground}
  \node[mqtt figure frame,fit=(figureTitle)(contentBounds)] {};
\end{pgfonlayer}
