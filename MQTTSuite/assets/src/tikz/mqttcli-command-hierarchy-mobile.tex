\node[mqtt config,wide,below=\MQGapS of sub] (pub)
  {\texttt{pub}: topic + message + retain};

\coordinate (bus-session) at ($(session.west)+(-7mm,0)$);

\draw[mqtt contains] (bus-pub) -- (pub.west);
