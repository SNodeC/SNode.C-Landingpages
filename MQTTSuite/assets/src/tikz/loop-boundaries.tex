\node[mqtt control,card] (pass)
  {\textbf{1. Immediate source exclusion}\\[2mm]
   \texttt{Bridge::publish()} forwards to every connected destination except the origin member.};
\node[mqtt warning,card,right=\MQHorizontalGap of pass] (cycle)
  {\textbf{2. Topology / subscription design}\\[2mm]
   A later cycle such as A → B → A is still possible when forwarded topics match another member's input subscriptions.};
\node[mqtt control,card,right=\MQHorizontalGap of cycle] (private)
  {\textbf{3. Private reflection suppression}\\[2mm]
   \texttt{loop\_prevention=true} requests the private SNode.C CONNECT protocol-level MSB for cooperating SNode.C MQTT peers.};

\node[mini,below=\MQGapM of pass,xshift=-20mm] (a) {Broker A\\origin};
\node[mqtt process,minimum width=29mm,text width=24mm,right=\MQHorizontalGapTight of a] (forward) {one forwarding pass};
\node[mini,right=\MQHorizontalGapTight of forward,yshift=8mm] (b) {Broker B};
\node[mini,right=\MQHorizontalGapTight of forward,yshift=-8mm] (c) {Broker C};
\draw[mqtt data flow] (a.east) -- (forward.west);
\draw[mqtt data flow] \MQPortEast{forward}{.68} -- ++(6mm,0) |- (b.west);
\draw[mqtt data flow] \MQPortEast{forward}{.32} -- ++(6mm,0) |- (c.west);
\node[mqtt note,minimum width=64mm,text width=58mm,below=\MQGapM of forward] (exclude)
  {A is not a destination in this pass. This does not prevent a later topology cycle.};

\node[mqtt note,minimum width=64mm,text width=58mm,below=\MQGapM of cycle] (operator)
  {Operator-owned controls: subscription partitioning, prefix design, and topology.};
\node[mqtt note,minimum width=64mm,text width=58mm,below=\MQGapM of private] (nonguarantee)
  {Non-standard MQTT extension; not a guarantee that arbitrary cyclic or third-party topologies are loop-safe.};
