\node[mqtt external,stage] (publish)
  {received MQTT PUBLISH\\topic + payload + QoS + flags};
\node[mqtt control,stage,right=\MQHorizontalGapLoose of publish] (store)
  {MQTTStore\\receive / store path};
\draw[mqtt data flow] (publish.east) -- (store.west);

\node[row,right=38mm of store,yshift=17mm] (raw)
  {raw-table INSERT attempt\\for every PUBLISH\\topic, payload, QoS, retain / DUP, packet id};
\node[mqtt decision,right=38mm of store,yshift=-17mm] (eligible)
  {valid JSON\\+ projection match?};

\draw[mqtt data flow] \MQPortEast{store}{.70} -- ++(8mm,0) |- node[mqtt edge label,near end,above]{1 · issue raw insert} (raw.west);
\draw[mqtt control flow] \MQPortEast{store}{.30} -- ++(8mm,0) |- node[mqtt edge label,near end,below]{2 · evaluate projections} (eligible.west);

\node[mqtt process,right=\MQHorizontalGapLoose of eligible,minimum width=50mm,text width=45mm] (projection)
  {optional projection extraction\\topic / JSON / literal values};
\node[row,right=\MQHorizontalGap of projection] (typed)
  {typed-row INSERT attempt\\zero / one / many writes};
\draw[mqtt control flow] (eligible.east) -- node[mqtt branch label,above]{YES} (projection.west);
\draw[mqtt data flow] (projection.east) -- node[mqtt edge label,above]{independent insert attempt(s)} (typed.west);

\node[mqtt note,below=\MQGapM of eligible,minimum width=49mm,text width=44mm] (skip)
  {NO: projection path is skipped};
\draw[mqtt association] (eligible.south) -- node[mqtt branch label,right]{NO} (skip.north);

\node[mqtt note,below=\MQGapM of raw,minimum width=49mm,text width=44mm] (independent)
  {raw insert success/failure does not gate projection processing};
\draw[mqtt association] (raw.south) -- (independent.north);

\begin{pgfonlayer}{background}
  \node[mqtt figure frame,fit=(publish)(store)(raw)(eligible)(projection)(typed)(skip)(independent)] (frame) {};
\end{pgfonlayer}
\node[mqtt section title] at ($(frame.north west)+(\MQTitleGap,-\MQTitleGap)$)
  {Raw envelope first, optional typed projections};
\node[mqtt annotation,anchor=east] at ($(frame.north east)+(-\MQTitleGap,-\MQTitleGap)$)
  {raw INSERT is issued before optional projection INSERTs; their outcomes are independent};
