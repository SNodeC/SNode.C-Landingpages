\begin{tikzpicture}[mqtt desktop]
\tikzset{
  main/.style={mqtt process,minimum width=36mm,text width=31mm,minimum height=14mm},
  source/.style={mqtt config,minimum width=25mm,text width=21mm,minimum height=16mm},
  endnote/.style={mqtt note,minimum width=24mm,text width=20mm,minimum height=11mm},
  qualifier/.style={mqtt note,minimum width=36mm,text width=31mm,minimum height=14mm}
}
\def\MQFlowSpan{8.3mm}
\def\MQHalfFlow{4.15mm}
\def\MQGroupLead{12.8mm}

\node[mqtt data,main] (publish) at (-22.15mm,0)
  {\textbf{MQTT PUBLISH}\\topic + payload + metadata};
\node[mqtt control,main] (store) at (22.15mm,0)
  {\textbf{MQTTStore}\\\texttt{store(message)}};
\draw[mqtt data flow] (publish.east) -- (store.west);

% Two independent paths leave store(message): raw INSERT and projection work.
\node[mqtt database,minimum width=36mm,below=\MQFlowSpan of store,xshift=-57mm] (raw)
  {raw-table\\INSERT attempt};
\node[mqtt decision,minimum width=31mm,below=\MQFlowSpan of store,xshift=3mm] (json)
  {valid JSON\\payload?};

\draw[mqtt data flow] \MQPortSouth{store}{.30}
  -- ++(0,-\MQHalfFlow) -| (raw.north);
\draw[mqtt control flow] \MQPortSouth{store}{.70}
  -- ++(0,-\MQHalfFlow) -| (json.north);

\node[qualifier,below=\MQGapS of raw] (rawNote)
  {Queued first. Raw completion does not gate projection processing.};
\draw[mqtt association] (raw.south) -- (rawNote.north);

\node[endnote,right=\MQFlowSpan of json] (notJson)
  {NO\\projection path ends};
\draw[mqtt control flow] (json.east) -- node[mqtt branch label,above]{NO} (notJson.west);

\node[mqtt decision,minimum width=34mm,below=\MQFlowSpan of json] (match)
  {any matching\\projection?};
\draw[mqtt control flow] (json.south) -- node[mqtt branch label,right]{YES} (match.north);

\node[endnote,right=\MQFlowSpan of match] (noMatch)
  {NO\\no typed INSERT};
\draw[mqtt control flow] (match.east) -- node[mqtt branch label,above]{NO} (noMatch.west);

% Exactly one configured source per SQL column; all three source forms are peers.
\node[source,below=\MQGroupLead of match] (pointer)
  {\texttt{json\_pointer}\\parsed JSON\\value};
\node[source,left=\MQHorizontalGapTight of pointer] (topic)
  {\texttt{topic\_level}\\zero-based\\topic segment};
\node[source,right=\MQHorizontalGapTight of pointer] (literal)
  {\texttt{literal}\\configured\\constant};

\node[fit=(topic)(pointer)(literal),inner sep=4.5mm] (sourceBounds) {};
\draw[mqtt control flow] (match.south)
  -- node[mqtt branch label,right]{YES · each match} (sourceBounds.north);

\node[mqtt database,minimum width=50mm,text width=44mm,below=\MQFlowSpan of sourceBounds] (typed)
  {typed projection INSERT attempt(s)\\one per matching projection};
\draw[mqtt data flow] (sourceBounds.south)
  -- node[mqtt edge label,right]{typed SQL column values} (typed.north);

\coordinate (leftEdge) at (publish.west);
\coordinate (titleY) at ($(publish.north)+(0,\MQTitleGap)$);
\node[mqtt section title,anchor=south west] (figureTitle)
  at (leftEdge |- titleY) {JSON/topic projection extraction};

\begin{pgfonlayer}{background}
  \node[mqtt figure frame,fit=(figureTitle)(publish)(store)(raw)(rawNote)(json)(notJson)(match)(noMatch)(sourceBounds)(typed)] {};
  \node[mqtt group,fit=(topic)(pointer)(literal),inner sep=4.5mm] (sources) {};
\end{pgfonlayer}
\node[mqtt edge label,anchor=west]
  at ($(sources.north west)+(3mm,0)$) {configured source for each SQL column};
