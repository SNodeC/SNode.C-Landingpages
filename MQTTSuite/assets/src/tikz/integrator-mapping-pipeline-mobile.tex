\begin{document}
\begin{tikzpicture}[mqtt mobile]
\tikzset{
  wide/.style={minimum width=72mm,text width=64mm},
  outputleft/.style={mqtt process,minimum width=37mm,text width=31mm},
  outputright/.style={mqtt process,minimum width=37mm,text width=31mm},
  sendleft/.style={mqtt process,minimum width=39mm,text width=33mm},
  queueright/.style={mqtt state,minimum width=39mm,text width=33mm},
  sendright/.style={mqtt process,minimum width=39mm,text width=33mm}
}

\def\MQArrowSpan{8.3mm}

% Subscription setup remains a separate centered semantic group.
\node[mqtt config,wide] (mappingdoc)
  {mapping document — subscription filters + \texttt{subscription.qos}};
\node[mqtt control,wide,below=\MQArrowSpan of mappingdoc] (extract)
  {\texttt{extractSubscriptions()} builds MQTT topic filters};
\node[mqtt config,wide,below=\MQArrowSpan of extract] (subscribe)
  {SUBSCRIBE setup — requested delivery QoS};
\draw[mqtt control flow] (mappingdoc.south) -- (extract.north);
\draw[mqtt control flow] (extract.south) -- (subscribe.north);

% Per-message runtime spine stays centered.
\node[mqtt external,wide,below=\MQLaneGap of subscribe] (brokerin)
  {Broker delivers PUBLISH selected by active subscriptions};
\node[mqtt process,wide,below=\MQArrowSpan of brokerin] (client)
  {MQTTIntegrator MQTT client};
\node[mqtt control,wide,below=\MQArrowSpan of client] (mapper)
  {\texttt{MqttMapper::getMappings()} — topic-tree first match};
\node[mqtt control,wide,below=\MQArrowSpan of mapper] (rules)
  {evaluate present \texttt{static}, \texttt{value}, and \texttt{json} keys independently};
\draw[mqtt data flow] (brokerin.south) -- (client.north);
\draw[mqtt data flow] (client.south) -- (mapper.north);
\draw[mqtt control flow] (mapper.south) -- (rules.north);

% Output siblings are centered as one pair with the canonical 8 mm edge gap.
\matrix (outputs) [
  matrix of nodes,
  inner sep=0pt,
  row sep=0pt,
  below=\MQArrowSpan of rules,
  column sep=\MQHorizontalGap,
  nodes={anchor=north}
] {
  |[outputleft,name=immediate]| {immediate outputs\\$0..n$ publications\\mapping QoS + retain} &
  |[outputright,name=scheduled]| {scheduled outputs\\$0..m$ publications\\delay + mapping QoS / retain} \\
};
\coordinate (rulesLeft) at ($(rules.south west)!0.333333!(rules.south east)$);
\coordinate (rulesRight) at ($(rules.south west)!0.666667!(rules.south east)$);
\coordinate (rulesOutputsMid) at ($(rules.south)!0.5!(immediate.north)$);
\draw[mqtt control flow]
  (rulesLeft) -- (rulesLeft |- rulesOutputsMid)
  -- (immediate.north |- rulesOutputsMid) -- (immediate.north);
\draw[mqtt control flow]
  (rulesRight) -- (rulesRight |- rulesOutputsMid)
  -- (scheduled.north |- rulesOutputsMid) -- (scheduled.north);

\node[sendleft,below=\MQArrowSpan of immediate] (sendnow)
  {\texttt{sendPublish(...)}\\immediately};
\node[queueright,below=\MQArrowSpan of scheduled] (queue)
  {enqueue in\\\texttt{DelayedQueue}};
\draw[mqtt data flow] (immediate.south) -- (sendnow.north);
\draw[mqtt control flow] (scheduled.south) -- (queue.north);

\node[sendright,below=\MQArrowSpan of queue] (sendlater)
  {when due:\\\texttt{sendPublish(...)}};
\draw[mqtt data flow] (queue.south) -- (sendlater.north);

% Note is centered on the main spine and remains explanatory only.
\coordinate (noteY) at ($(sendlater.south)+(0,-\MQGapM)$);
\node[mqtt note,wide,anchor=north] (note) at (rules.center |- noteY)
  {Both republish paths use the same MQTT client connection.\\Outgoing PUBLISH QoS is independent of SUBSCRIBE QoS.};

% Setup label uses canonical clearance above the first setup box.
\node[mqtt annotation,anchor=south west] (setupLabel)
  at ($(mappingdoc.north west)+(0,\MQLabelGap)$)
  {subscription setup};

% Invisible relative bounds let the figure title and outer frame be derived from
% the semantic setup group without hand-positioning either one.
\node[draw=none,fill=none,inner sep=\MQGroupPad,fit=(setupLabel)(mappingdoc)(extract)(subscribe)]
  (setupBounds) {};
\node[mqtt section title,anchor=south west] (figureTitle)
  at ($(setupBounds.north west)+(0,\MQTitleGap)$)
  {MQTTIntegrator mapping pipeline};

% Outer frame first; semantic setup group afterwards. Fitting the title into the
% frame provides canonical clearance from the frame's top border.
\begin{pgfonlayer}{background}
  \node[mqtt figure frame,fit=(figureTitle)(setupBounds)(brokerin)(client)(mapper)(rules)(outputs)(sendnow)(queue)(sendlater)(note)] (frame) {};
  \node[mqtt group,fit=(setupLabel)(mappingdoc)(extract)(subscribe)] (setupframe) {};
\end{pgfonlayer}
\end{tikzpicture}
\end{document}
