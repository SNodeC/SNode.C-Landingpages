\begin{tikzpicture}[mqtt desktop]
\tikzset{
  setup/.style={mqtt config,minimum width=35mm,text width=29mm},
  extractstage/.style={mqtt control,minimum width=43mm,text width=37mm},
  runtime/.style={mqtt process,minimum width=35mm,text width=29mm},
  externalstage/.style={mqtt external,minimum width=35mm,text width=29mm},
  mapperstage/.style={mqtt control,minimum width=48mm,text width=42mm},
  controlstage/.style={mqtt control,minimum width=42mm,text width=36mm},
  outputleft/.style={mqtt process,minimum width=36mm,text width=30mm},
  outputright/.style={mqtt process,minimum width=39mm,text width=33mm},
  sendstage/.style={mqtt process,minimum width=40mm,text width=34mm},
  queuestage/.style={mqtt state,minimum width=39mm,text width=33mm},
  sharednote/.style={mqtt note,minimum width=82mm,text width=76mm}
}

\def\MQArrowSpan{8.3mm}

% Subscription setup: one centered semantic band.
\matrix (setuprow) [
  matrix of nodes,
  inner sep=0pt,
  row sep=0pt,
  column sep=\MQArrowSpan,
  nodes={anchor=center}
] {
  |[setup,name=mappingdoc]| {mapping document\\subscription filters + \texttt{subscription.qos}} &
  |[extractstage,name=extract]| {\texttt{extractSubscriptions()}\\build MQTT topic filters} &
  |[setup,name=subscribe]| {SUBSCRIBE setup\\requested delivery QoS} \\
};
\draw[mqtt control flow] (mappingdoc.east) -- (extract.west);
\draw[mqtt control flow] (extract.east) -- (subscribe.west);

% Per-message runtime row: centered as one unit below setup.
\matrix (runtimeRow) [
  matrix of nodes,
  inner sep=0pt,
  row sep=0pt,
  below=\MQLaneGap of setuprow,
  column sep=\MQArrowSpan,
  nodes={anchor=center}
] {
  |[externalstage,name=brokerin]| {Broker delivers PUBLISH\\selected by active subscriptions} &
  |[runtime,name=client]| {MQTTIntegrator\\MQTT client} &
  |[mapperstage,name=mapper]| {\texttt{MqttMapper::getMappings()}\\topic-tree first match} \\
};
\draw[mqtt data flow] (brokerin.east) -- (client.west);
\draw[mqtt data flow] (client.east) -- (mapper.west);

% Mapping evaluation is centered on the complete figure/runtime row.
\coordinate (rulesNorthY) at ($(mapper.south)+(0,-\MQArrowSpan)$);
\node[controlstage,anchor=north] (rules) at (runtimeRow.center |- rulesNorthY)
  {evaluate present mapping keys\\\texttt{static}, \texttt{value}, \texttt{json} independently};

% Mapper -> rules: canonical centered dogleg; total vertical span = one arrow span.
\coordinate (mapperRulesMid) at ($(mapper.south)!0.5!(rules.north)$);
\draw[mqtt control flow]
  (mapper.south) -- (mapper.south |- mapperRulesMid)
  -- (rules.north |- mapperRulesMid) -- (rules.north);

% Independent output collections are a centered sibling pair below rules.
\matrix (outputs) [
  matrix of nodes,
  inner sep=0pt,
  row sep=0pt,
  below=\MQArrowSpan of rules,
  column sep=\MQHorizontalGap,
  nodes={anchor=north}
] {
  |[outputleft,name=immediate]| {immediate outputs\\$0..n$ publications\\mapping-defined QoS + retain} &
  |[outputright,name=scheduled]| {scheduled outputs\\$0..m$ publications\\delay + mapping-defined QoS / retain} \\
};
\coordinate (rulesLeft) at ($(rules.south west)!0.333333!(rules.south east)$);
\coordinate (rulesRight) at ($(rules.south west)!0.666667!(rules.south east)$);
\coordinate (rulesOutputsMid) at ($(rules.south)!0.5!(immediate.north)$);
\draw[mqtt control flow]
  (rulesLeft) -- (rulesLeft |- rulesOutputsMid)
  -- (immediate.north |- rulesOutputsMid) -- (immediate.north);
\draw[mqtt control flow]
  (rulesRight) -- (rulesRight |- rulesOutputsMid)
  -- (scheduled.north |- rulesOutputsMid) -- (scheduled.north);

% Immediate and delayed terminal paths keep their real unequal semantic depth.
\node[sendstage,below=\MQArrowSpan of immediate] (sendnow)
  {\texttt{sendPublish(...)}\\immediately};
\node[queuestage,below=\MQArrowSpan of scheduled] (queue)
  {enqueue in \texttt{DelayedQueue}\\timer releases when due};
\draw[mqtt data flow] (immediate.south) -- (sendnow.north);
\draw[mqtt control flow] (scheduled.south) -- (queue.north);

\node[sendstage,below=\MQArrowSpan of queue] (sendlater)
  {when due: \texttt{sendPublish(...)} };
\draw[mqtt data flow] (queue.south) -- (sendlater.north);

% Shared explanatory note: centered on the figure axis, below the deepest path.
\coordinate (noteY) at ($(sendlater.south)+(0,-\MQGapM)$);
\node[sharednote,anchor=north] (note) at (rules.center |- noteY)
  {Both republish paths use the same MQTT client connection.\\Outgoing PUBLISH QoS is an output-mapping decision, independent of SUBSCRIBE QoS.};

% Setup label uses canonical clearance above the first setup box.
\node[mqtt annotation,anchor=south west] (setupLabel)
  at ($(mappingdoc.north west)+(0,\MQLabelGap)$)
  {subscription setup --- outside the per-PUBLISH processing spine};

% Invisible relative bounds let the figure title and outer frame be derived from
% the semantic setup group without hand-positioning either one.
\node[draw=none,fill=none,inner sep=\MQGroupPad,fit=(setupLabel)(mappingdoc)(extract)(subscribe)]
  (setupBounds) {};
\node[mqtt section title,anchor=south west] (figureTitle)
  at ($(setupBounds.north west)+(0,\MQTitleGap)$)
  {MQTTIntegrator mapping pipeline};

% Outer frame is painted first. The setup group is painted afterwards so its
% border remains visible. Fitting the title gives canonical frame clearance.
\begin{pgfonlayer}{background}
  \node[mqtt figure frame,fit=(figureTitle)(setupBounds)(runtimeRow)(rules)(outputs)(sendnow)(queue)(sendlater)(note)] (frame) {};
  \node[mqtt group,fit=(setupLabel)(mappingdoc)(extract)(subscribe)] (setupframe) {};
\end{pgfonlayer}

% Canonical SNode.C desktop canvas: 160 mm, without scaling the drawing.
\coordinate (canvasCenter) at (current bounding box.center);
\coordinate (canvasSW) at ($(canvasCenter)+(-80mm,0)$ |- current bounding box.south);
\coordinate (canvasNE) at ($(canvasCenter)+(80mm,0)$ |- current bounding box.north);
\path[use as bounding box]
  ([yshift=-7pt]canvasSW) rectangle ([yshift=7pt]canvasNE);
