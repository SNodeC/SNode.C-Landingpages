\begin{tikzpicture}[mqtt desktop]
\tikzset{
  src/.style={mqtt config,minimum width=32mm,text width=27mm},
  main/.style={mqtt control,minimum width=44mm,text width=38mm},
  outcome/.style={mqtt process,minimum width=36mm,text width=31mm},
  doc/.style={mqtt config,minimum width=39mm,text width=34mm}
}

\node[src] (defaults) {API / compiled defaults\\lowest precedence};
\node[src,below=\MQGapS of defaults] (file) {SNode.C configuration file};
\node[src,below=\MQGapS of file] (cli) {command line\\highest precedence};
\node[mqtt annotation,above=3mm of defaults] (precedence)
  {precedence increases: defaults $\rightarrow$ file $\rightarrow$ command line};

\node[main,right=\MQHorizontalGap of file] (effective)
  {effective parsed SNode.C configuration\\application $\rightarrow$ named instance\\$\rightarrow$ sections / options};

% Exact symmetric H-V-H doglegs: the turn x-coordinate is the midpoint
% between the source and destination border ports, so the two horizontal outer
% legs have exactly equal length.
\coordinate (effective-defaults) at \MQPortWest{effective}{.75};
\coordinate (effective-cli) at \MQPortWest{effective}{.25};
\coordinate (defaults-mid) at ($(defaults.east)!0.5!(effective-defaults)$);
\coordinate (cli-mid) at ($(cli.east)!0.5!(effective-cli)$);
\draw[mqtt control flow] (defaults.east) -- (defaults.east -| defaults-mid) |- (effective-defaults);
\draw[mqtt control flow] (file.east) -- \MQPortWest{effective}{.50};
\draw[mqtt control flow] (cli.east) -- (cli.east -| cli-mid) |- (effective-cli);

\node[outcome,right=\MQHorizontalGapLoose of effective,yshift=12mm] (inspect)
  {inspect effective state\\\texttt{--show-config}\\{\MQCodeSmallFont --command-line=active}};
\node[outcome,right=\MQHorizontalGapLoose of effective,yshift=-12mm] (write)
  {serialize effective parsed state\\\texttt{--write-config}};
\node[outcome,below=\MQGapS of write] (saved)
  {written application configuration file\\\textbf{not runtime success}};

\coordinate (effective-inspect) at \MQPortEast{effective}{.70};
\coordinate (effective-write) at \MQPortEast{effective}{.30};
\coordinate (inspect-mid) at ($(effective-inspect)!0.5!(inspect.west)$);
\coordinate (write-mid) at ($(effective-write)!0.5!(write.west)$);
\draw[mqtt observation] (effective-inspect) -- (effective-inspect -| inspect-mid) |- (inspect.west);
\draw[mqtt control flow] (effective-write) -- (effective-write -| write-mid) |- (write.west);
\draw[mqtt control flow] (write.south) -- (saved.north);

\coordinate (apptitleguard) at ($(precedence.north)+(0,\MQGapL)$);
\begin{pgfonlayer}{background}
  \node[mqtt figure frame,fit=(apptitleguard)(precedence)(defaults)(file)(cli)(effective)(inspect)(write)(saved)] (appframe) {};
\end{pgfonlayer}
\node[mqtt section title] at ($(appframe.north west)+(\MQTitleGap,-\MQTitleGap)$)
  {SNode.C application configuration};

% The domain-document panel owns its own title band. Keep enough external gap
% that the band's fit/padding cannot intrude into the application frame.
\node[doc,below=22mm of appframe] (bridge)
  {bridge-definition JSON\\MQTTBridge};
\node[doc,left=\MQHorizontalGap of bridge] (mapping)
  {mapping JSON\\Integrator / optional Broker mapper};
\node[doc,right=\MQHorizontalGap of bridge] (projection)
  {projection JSON\\MQTTStore};
\node[mqtt note,below=\MQGapS of bridge,minimum width=86mm,text width=80mm] (separate)
  {separate domain documents with separate schemas / lifecycles;\\not merged by SNode.C precedence};

\coordinate (domaintitleguard) at ($(mapping.north)+(0,\MQGapL)$);
\begin{pgfonlayer}{background}
  \node[mqtt group,fit=(domaintitleguard)(mapping)(bridge)(projection)(separate)] (domainframe) {};
\end{pgfonlayer}
\node[mqtt section title] at ($(domainframe.north west)+(\MQTitleGap,-\MQTitleGap)$)
  {Application-specific domain documents};

\end{tikzpicture}
