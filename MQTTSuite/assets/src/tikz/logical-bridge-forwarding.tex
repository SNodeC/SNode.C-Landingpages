\begin{tikzpicture}[mqtt desktop]
\tikzset{
  broker/.style={mqtt external,minimum width=30mm,text width=25mm},
  member/.style={mqtt application,minimum width=37mm,text width=31mm},
  core/.style={mqtt control,minimum width=41mm,text width=35mm},
  peer/.style={mqtt application,minimum width=32mm,text width=27mm},
  peerbroker/.style={mqtt external,minimum width=32mm,text width=27mm}
}

\node[broker] (a) {Broker A\\source domain};
\node[member,right=\MQHorizontalGapTight of a] (ma)
  {member A\\configured subscriptions\\select received PUBLISH};
\node[core,right=\MQHorizontalGapTight of ma] (bridge)
  {logical bridge\\fan-out forwarding};

\draw[mqtt data flow] (a.east) -- (ma.west);
\draw[mqtt data flow] (ma.east) -- (bridge.west);

\node[mqtt state,below=\MQGapM of bridge,minimum width=74mm,text width=68mm] (fanout)
  {connected destination set\\all currently connected bridge members\\except origin member A};
\draw[mqtt data flow] (bridge.south) -- (fanout.north);

\node[peer,below=\MQGapM of fanout,xshift=-21mm] (mb) {member B\\destination};
\node[peer,below=\MQGapM of fanout,xshift=21mm] (mc) {member C\\destination};
\node[peerbroker,below=\MQGapS of mb] (b) {Broker B};
\node[peerbroker,below=\MQGapS of mc] (c) {Broker C};

% Two evenly distributed fan-out ports. Each V-H-V connector turns exactly at
% the midpoint between source and destination border levels, so both vertical
% outer legs are equal and the connector remains orthogonal at both boxes.
\coordinate (fanoutB) at ($(fanout.south west)!.333333!(fanout.south east)$);
\coordinate (fanoutC) at ($(fanout.south west)!.666667!(fanout.south east)$);
\coordinate (fanoutMidB) at ($(fanoutB)!0.5!(mb.north)$);
\coordinate (fanoutMidC) at ($(fanoutC)!0.5!(mc.north)$);
\draw[mqtt data flow] (fanoutB) -- (fanoutB |- fanoutMidB) -| (mb.north);
\draw[mqtt data flow] (fanoutC) -- (fanoutC |- fanoutMidC) -| (mc.north);
\draw[mqtt data flow] (mb.south) -- (b.north);
\draw[mqtt data flow] (mc.south) -- (c.north);

\coordinate (brokersMid) at ($(b.south)!0.5!(c.south)$);
\node[mqtt note,below=\MQGapM of brokersMid,minimum width=94mm,text width=88mm] (preserve)
  {message bytes, received QoS, and retain are preserved;\\the forwarded topic is rebuilt from bridge/member prefixes and documented separately};

\coordinate (titleguard) at ($(ma.north)+(0,\MQGapL)$);
\begin{pgfonlayer}{background}
  \node[mqtt figure frame,fit=(titleguard)(a)(ma)(bridge)(fanout)(mb)(mc)(b)(c)(preserve)] (frame) {};
\end{pgfonlayer}
\node[mqtt section title] at ($(frame.north west)+(\MQTitleGap,-\MQTitleGap)$)
  {Logical bridge forwarding};
