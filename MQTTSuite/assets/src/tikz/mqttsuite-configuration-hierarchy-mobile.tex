\documentclass[tikz,border=0pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage{lmodern}

\begin{document}
\begin{tikzpicture}[mqtt mobile]
\tikzset{
  root/.style={minimum width=70mm,text width=62mm,minimum height=14mm},
  inst/.style={minimum width=70mm,text width=62mm,minimum height=14mm},
  peer/.style={mqtt config,minimum width=30mm,text width=25mm,minimum height=22mm},
  wideNote/.style={mqtt note,minimum width=70mm,text width=62mm,minimum height=16mm}
}

\node[mqtt application,root] (app)
  {\textbf{application root}\\e.g. \texttt{mqttcli}};
\node[mqtt control,inst,below=\MQGapM of app] (instance)
  {\textbf{named connection / listener instance}\\e.g. \texttt{in-mqtt}};
\draw[mqtt contains] (app.south) -- (instance.north);

\node[mqtt annotation,below=\MQLabelGap of instance] (sectionLabel)
  {selected instance — peer sections};
\matrix (peerGrid) [
  matrix of nodes,
  below=\MQGapS of sectionLabel,
  row sep=\MQGapS,
  column sep=\MQHorizontalGapTight,
  nodes={peer,anchor=center}
] {
  {\textbf{endpoint + transport}\\\texttt{remote} / \texttt{local}\\address + socket / TLS} &
  {\textbf{HTTP / WebSocket}\\optional\\instance-family dependent} \\
  {\textbf{MQTT session}\\CONNECT / session settings\\when MQTT is used} &
  {\textbf{app-specific}\\e.g. \texttt{sub}, \texttt{pub}\\or \texttt{db} with nested\\\texttt{storage}} \\
};
\node[fit=(instance)(sectionLabel)(peerGrid),inner sep=\MQGroupPad] (instanceBounds) {};

\node[wideNote,below=\MQGapS of instanceBounds] (ownership)
  {Peer section families belong to the selected instance. Exact sets vary. This is configuration ownership, not runtime protocol layering or execution order.};

\node[mqtt annotation,below=\MQGapM of ownership] (surfaceLabel)
  {configuration surfaces};
\node[wideNote,below=\MQLabelGap of surfaceLabel] (values)
  {\textbf{value precedence}\\API / defaults $<$ configuration file $<$ command line};
\node[wideNote,below=\MQGapS of values] (inspect)
  {\textbf{introspection}\\tree: \texttt{--help=expanded}\\state: \texttt{--show-config}\\\texttt{--command-line=...}};
\node[fit=(surfaceLabel)(values)(inspect),inner sep=\MQGroupPad] (surfaceBounds) {};

\coordinate (titleY) at ($(app.north)+(0,\MQTitleGap)$);
\node[mqtt section title,anchor=south west] (figureTitle)
  at ($(instanceBounds.west |- titleY)$) {Configuration ownership hierarchy};
\begin{pgfonlayer}{background}
  \node[mqtt figure frame,fit=(figureTitle)(app)(instanceBounds)(ownership)(surfaceBounds)] {};
  \node[mqtt group,fit=(instance)(sectionLabel)(peerGrid),inner sep=\MQGroupPad] {};
  \node[mqtt group,fit=(surfaceLabel)(values)(inspect),inner sep=\MQGroupPad] {};
\end{pgfonlayer}
\end{tikzpicture}
\end{document}
