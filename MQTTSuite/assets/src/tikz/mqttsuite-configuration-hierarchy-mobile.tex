\tikzset{wide/.style={minimum width=72mm,text width=64mm}}

\node[mqtt application,wide] (app) {application root};
\node[mqtt control,wide,below=\MQGapS of app] (instance)
  {named connection / listener instance};
\draw[mqtt contains] (app.south) -- (instance.north);

\node[mqtt config,wide,below=\MQGapM of instance] (network)
  {network / endpoint sections — address + transport};
\node[mqtt config,wide,below=\MQGapS of network] (http)
  {optional HTTP / WebSocket / TLS-related sections when used};
\node[mqtt config,wide,below=\MQGapS of http] (session)
  {MQTT session section where the application uses MQTT};
\node[mqtt config,wide,below=\MQGapS of session] (appsection)
  {application-specific section(s): e.g. \texttt{sub}, \texttt{pub}, storage, mapper options};

\coordinate (bus-network) at ($(network.west)+(-7mm,0)$);
\coordinate (bus-http) at ($(http.west)+(-7mm,0)$);
\coordinate (bus-session) at ($(session.west)+(-7mm,0)$);
\coordinate (bus-app) at ($(appsection.west)+(-7mm,0)$);
\draw[mqtt contains] (bus-network) -- (bus-app);
\draw[mqtt contains] (instance.west) -- ++(-7mm,0) |- (bus-session);
\draw[mqtt contains] (bus-network) -- (network.west);
\draw[mqtt contains] (bus-http) -- (http.west);
\draw[mqtt contains] (bus-session) -- (session.west);
\draw[mqtt contains] (bus-app) -- (appsection.west);

\node[mqtt note,wide,below=\MQGapM of appsection] (ownership)
  {Peer sections are owned by the instance; runtime protocol layering/order is separate. Exact section sets differ by application and instance family.};
\node[mqtt note,wide,below=\MQGapS of ownership] (surfaces)
  {API/default $<$ configuration file $<$ command line; supported inspection surfaces expose the parsed hierarchy.};

\begin{pgfonlayer}{background}
  \node[mqtt figure frame,fit=(app)(instance)(network)(http)(session)(appsection)(ownership)(surfaces)] (frame) {};
\end{pgfonlayer}
\node[mqtt section title] at ($(frame.north west)+(\MQTitleGap,-\MQTitleGap)$)
  {Configuration ownership hierarchy};
