\tikzset{

\node[mqtt annotation] (precedence)
  {precedence: defaults $<$ file $<$ command line};
\node[mqtt config,minimum width=24mm,text width=18mm,below=3mm of precedence] (file)
  {config\\file};
\node[mqtt config,minimum width=24mm,text width=18mm,left=\MQGapS of file] (defaults)
  {API / compiled\\defaults};
\node[mqtt config,minimum width=24mm,text width=18mm,right=\MQGapS of file] (cli)
  {command\\line};

\node[mqtt control,wide,below=\MQBranchGap of file] (effective)
  {effective parsed SNode.C application / instance configuration};

% Three configuration sources converge directly on the effective parsed state.
% The two off-axis V-H-V connectors turn at the exact midpoint between source
% and destination border levels, making their outer vertical legs equal.
\coordinate (effectiveDefaults) at ($(effective.north west)!.25!(effective.north east)$);
\coordinate (effectiveFile) at ($(effective.north west)!.50!(effective.north east)$);
\coordinate (effectiveCli) at ($(effective.north west)!.75!(effective.north east)$);
\coordinate (defaultsMid) at ($(defaults.south)!0.5!(effectiveDefaults)$);
\coordinate (cliMid) at ($(cli.south)!0.5!(effectiveCli)$);
\draw[mqtt control flow] (defaults.south) -- (defaults.south |- defaultsMid) -| (effectiveDefaults);
\draw[mqtt control flow] (file.south) -- (effectiveFile);
\draw[mqtt control flow] (cli.south) -- (cli.south |- cliMid) -| (effectiveCli);

\node[mqtt process,inspectbox,below=\MQBranchGap of effective,xshift=-21mm] (inspect)
  {inspect\\\texttt{--show-config}\\{\MQCodeSmallFont\texttt{--command-line=active}}};
\node[mqtt process,half,below=\MQBranchGap of effective,xshift=21mm] (write)
  {serialize\\effective state\\\texttt{--write-config}};

\coordinate (effectiveInspect) at ($(effective.south west)!.32!(effective.south east)$);
\coordinate (effectiveWrite) at ($(effective.south west)!.68!(effective.south east)$);
\coordinate (inspectMid) at ($(effectiveInspect)!0.5!(inspect.north)$);
\coordinate (writeMid) at ($(effectiveWrite)!0.5!(write.north)$);
\draw[mqtt observation] (effectiveInspect) -- (effectiveInspect |- inspectMid) -| (inspect.north);
\draw[mqtt control flow] (effectiveWrite) -- (effectiveWrite |- writeMid) -| (write.north);

\node[mqtt config,half,below=\MQGapS of write] (saved)
  {written config file\\\textbf{not runtime success}};
\draw[mqtt control flow] (write.south) -- (saved.north);

\coordinate (apptitleguard) at ($(precedence.north)+(0,\MQGapL)$);
\begin{pgfonlayer}{background}
  \node[mqtt figure frame,fit=(apptitleguard)(precedence)(defaults)(file)(cli)(effective)(inspect)(write)(saved)] (appframe) {};
\end{pgfonlayer}
\node[mqtt section title] at ($(appframe.north west)+(\MQTitleGap,-\MQTitleGap)$)
  {Configuration hierarchy and persistence};

\node[mqtt config,wide,below=22mm of appframe] (mapping)
  {mapping JSON — Integrator / optional Broker mapper};
\node[mqtt config,wide,below=\MQGapS of mapping] (bridge)
  {bridge-definition JSON — MQTTBridge};
\node[mqtt config,wide,below=\MQGapS of bridge] (projection)
  {projection JSON — MQTTStore};
\node[mqtt note,wide,below=\MQGapS of projection] (note)
  {separate domain documents with separate schemas / lifecycles;\\not merged into the SNode.C precedence chain};

\coordinate (domaintitleguard) at ($(mapping.north)+(0,\MQGapL)$);
\begin{pgfonlayer}{background}
  \node[mqtt group,fit=(domaintitleguard)(mapping)(bridge)(projection)(note)] (domainframe) {};
\end{pgfonlayer}
\node[mqtt section title] at ($(domainframe.north west)+(\MQTitleGap,-\MQTitleGap)$)
  {Separate domain documents};
